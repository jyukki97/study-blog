{{- if and (not .IsHome) .IsSection -}} {{- /* .Pages는 바로 아래 단계만 포함한다. /learning/deep-dive/처럼 중첩 아카이브를 둔 섹션에서는 CollectionPage의 항목 수와 최신 목록이 실제 글보다 작아질 수 있으므로, 하위 정규 글까지 재귀적으로 수집한다. 목록 화면의 페이지네이션과 독립된 schema 전용 집계라 표시 결과는 바꾸지 않는다. */ -}} {{- $pages := .RegularPagesRecursive -}} {{- $elements := slice -}} {{- range $index, $page := first 12 ($pages.ByDate.Reverse) -}} {{- $elements = $elements | append (dict "@type" "ListItem" "position" (add $index 1) "url" $page.Permalink "name" $page.Title "description" ($page.Description | default ($page.Summary | plainify | truncate 110)) ) -}} {{- end -}} {{- $canonical := partial "canonical_url.html" . -}} {{- $collection := dict "@context" "https://schema.org" "@type" "CollectionPage" "name" .Title "description" (.Description | default (.Summary | plainify) | default .Site.Title) "url" $canonical "inLanguage" (.Site.Language.Lang | default "ko-kr") "isPartOf" (dict "@type" "WebSite" "name" .Site.Title "url" .Site.BaseURL) "about" (dict "@type" "Thing" "name" .Title) -}} {{- if gt (len $pages) 0 -}} {{- $collection = merge $collection (dict "numberOfItems" (len $pages)) -}} {{- end -}} {{- if gt (len $elements) 0 -}} {{- $collection = merge $collection (dict "mainEntity" (dict "@type" "ItemList" "itemListElement" $elements)) -}} {{- end -}} {{- end -}}
