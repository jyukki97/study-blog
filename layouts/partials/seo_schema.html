{{- if .IsPage -}} {{- /* 학습 아카이브는 일반 Article보다 기술 문서 성격이 강하다. TechArticle을 사용하면 검색 엔진이 블로그 소식과 실무 학습 자료를 구분하고, 모듈·태그 기반의 주제 신호를 더 일관되게 해석할 수 있다. */ -}} {{- $schemaType := "Article" -}} {{- if eq .Section "posts" -}} {{- $schemaType = "BlogPosting" -}} {{- else if eq .Section "learning" -}} {{- $schemaType = "TechArticle" -}} {{- end -}} {{- $description := .Description | default .Summary | plainify -}} {{- $keywords := .Params.keywords | default .Params.tags -}} {{- /* canonical 태그와 JSON-LD가 서로 다른 URL을 가리키면 검색 엔진의 대표 URL 신호가 분산될 수 있다. */ -}} {{- $canonical := partial "canonical_url.html" . -}} {{- /* 기존 글에는 category가 없는 경우가 있으므로, Article의 주제 신호가 사라지지 않게 섹션을 최소 fallback으로 남긴다. */ -}} {{- $articleSection := .Section -}} {{- with .Params.categories -}} {{- if reflect.IsSlice . -}} {{- $articleSection = delimit . ", " -}} {{- else -}} {{- $articleSection = . -}} {{- end -}} {{- end -}} {{- $authorName := .Site.Params.author | default .Site.Title -}} {{- $publisher := dict "@type" "Organization" "name" .Site.Title -}} {{- with .Site.Params.label.icon -}} {{- $publisher = merge $publisher (dict "logo" (dict "@type" "ImageObject" "url" (. | absURL))) -}} {{- end -}} {{- $schema := dict "@context" "https://schema.org" "@type" $schemaType "headline" .Title "description" $description "mainEntityOfPage" (dict "@type" "WebPage" "@id" $canonical) "url" $canonical "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "inLanguage" (.Site.Language.Lang | default "ko-kr") "wordCount" .WordCount "author" (dict "@type" "Person" "name" $authorName) "publisher" $publisher -}} {{- if $keywords -}} {{- $schema = merge $schema (dict "keywords" $keywords) -}} {{- end -}} {{- if $articleSection -}} {{- $schema = merge $schema (dict "articleSection" $articleSection) -}} {{- end -}} {{- /* 개별 커버가 없는 글도 검색 결과에서 일관된 대표 이미지를 갖도록 한다. social_meta와 같은 PNG를 사용해 SVG를 처리하지 못하는 crawler의 누락을 피한다. */ -}} {{- $schemaImage := "images/logo.png" | absURL -}} {{- with .Params.cover.image -}} {{- $schemaImage = . | absURL -}} {{- end -}} {{- $schema = merge $schema (dict "image" (slice $schemaImage)) -}} {{- with .Params.tags -}} {{- $about := slice -}} {{- range . -}} {{- $about = $about | append (dict "@type" "Thing" "name" .) -}} {{- end -}} {{- $schema = merge $schema (dict "about" $about) -}} {{- end -}} {{- with .Params.series -}} {{- $seriesName := . -}} {{- if reflect.IsSlice . -}} {{- $seriesName = delimit . ", " -}} {{- end -}} {{- $schema = merge $schema (dict "isPartOf" (dict "@type" "CreativeWorkSeries" "name" $seriesName)) -}} {{- end -}} {{- with .Params.learning_refs -}} {{- $mentions := slice -}} {{- range . -}} {{- $refURL := "" -}} {{- with .href -}}{{- $refURL = . | absURL -}}{{- end -}} {{- if and .title $refURL -}} {{- $mention := dict "@type" "CreativeWork" "name" .title "url" $refURL -}} {{- with .description -}} {{- $mention = merge $mention (dict "description" .) -}} {{- end -}} {{- $mentions = $mentions | append $mention -}} {{- end -}} {{- end -}} {{- if gt (len $mentions) 0 -}} {{- $schema = merge $schema (dict "mentions" $mentions) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 홈과 섹션에도 페이지 성격을 명시해 글 목록을 개별 Article로 오인하지 않게 한다. */ -}} {{- if .IsHome -}} {{- $siteDescription := site.Params.description | default site.Title | plainify -}} {{- $website := dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "url" ("/" | absURL) "description" $siteDescription "inLanguage" (site.Language.Lang | default "ko-KR") "publisher" (dict "@type" "Organization" "name" site.Title) -}} {{- end -}}
